\section{Inclusión Tecnológica}
\label{sec:inclusion}

\begin{itemize}
    \item \textbf{Democratización del testing automatizado}:
    \begin{itemize}
        \item Reducción de la barrera de entrada para PYMES y desarrolladores independientes
        \item Sistema accesible vía extensión VSCode (sin requerir infraestructura costosa)
        \item Documentación traducida a español para comunidades latinoamericanas
    \end{itemize}
    
    \item \textbf{Ejemplo cuantificado}:
    \begin{itemize}
        \item Costo tradicional: \$15,000 USD/año en herramientas empresariales
        \item Nuestra solución: \$200 USD/año (modelo open-core)
    \end{itemize}
\end{itemize}

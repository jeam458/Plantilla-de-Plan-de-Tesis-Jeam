\section{Detalle de Actividades}
\begin{itemize}
    \item \textbf{Fase 1: Preparación (Semanas 1-4)}
    \begin{itemize}
        \item Identificación de 50+ repositorios Ionic con pruebas unitarias
        \item Definición del dataset para fine-tuning (1,200+ componentes)
        \item Establecimiento de métricas base (cobertura, complejidad)
    \end{itemize}
    
    \item \textbf{Fase 2: Desarrollo (Semanas 5-12)}
    \begin{itemize}
        \item Implementación en Python del analizador de componentes
        \item Entrenamiento del modelo con casos Ionic específicos
        \item Integración con GitHub Actions y GitLab CI
    \end{itemize}
    
    \item \textbf{Fase 3: Validación (Semanas 10-15)}
    \begin{itemize}
        \item Pruebas A/B en 3 proyectos reales
        \item Encuesta a 15 desarrolladores Ionic
        \item Análisis estadístico de resultados
    \end{itemize}
    
    \item \textbf{Fase 4: Cierre (Semanas 14-16)}
    \begin{itemize}
        \item Documentación técnica y académica
        \item Preparación de materiales de difusión
    \end{itemize}
\end{itemize}

\section{Impacto Social Esperado}
\begin{itemize}

    \begin{itemize}
        \item 500+ desarrolladores Ionic en Latinoamérica (estimación basada en GitHub stats)
        \item 30+ startups tecnológicas de la región

    \item \textbf{Sectores prioritarios}:

    \end{itemize}
\end{itemize}
